    function processFile(file, callback) {

      var r = new FileReader();

      r.onload = function(e) { var d = new Uint8Array(e.target.result), wb = XLSX.read(d, {type:"array"}), csv = XLSX.utils.sheet_to_csv(wb.Sheets[wb.SheetNames[0]], {raw:false, dateNF:"DD-MM-YYYY"}); callback(csv); };

      r.readAsArrayBuffer(file);

    }


    function uploadData() {

      var fSPW = document.getElementById("fileSPW").files[0];

      var fMSR = document.getElementById("fileMSR").files[0];

      var fSTOCK = document.getElementById("fileSTOCK").files[0];

      if (!fSPW && !fMSR && !fSTOCK) { alert("Pilih minimal satu file untuk diupload!"); return; }

      

      document.getElementById("btnUpload").disabled = true; document.getElementById("btnUpload").innerText = "⏳ Memproses Data...";

      var spwT = "", msrT = "", stockT = "";


      function sendAll() {

        fetch(API_URL, {

          method: "POST",

          headers: { "Content-Type": "text/plain;charset=utf-8" },

          body: JSON.stringify({ action: "uploadData", spw: spwT, msr: msrT, stock: stockT })

        }).then(res => res.json()).then(res => {

          document.getElementById("status").innerText = res.message; document.getElementById("btnUpload").disabled = false; 

          document.getElementById("btnUpload").innerText = "Sinkronisasi Data"; dashboardData = []; sohDataGlobal = [];

        }).catch(err => {

          document.getElementById("status").innerText = "❌ Gagal koneksi ke server."; document.getElementById("btnUpload").disabled = false;

        });

      }


      var tasks = [];

      if (fSPW) tasks.push(function(next) { processFile(fSPW, function(c) { spwT = c; next(); }); });

      if (fMSR) tasks.push(function(next) { processFile(fMSR, function(c) { msrT = c; next(); }); });

      if (fSTOCK) tasks.push(function(next) { processFile(fSTOCK, function(c) { stockT = c; next(); }); });


      var step = 0;

      function runNext() { if (step < tasks.length) { tasks[step++](runNext); } else { sendAll(); } }

      runNext();

    }

    function gantiPin() {

      const oldPin = document.getElementById("oldPin").value, newPin = document.getElementById("newPin").value;

      const statusBox = document.getElementById("pinStatus");

      statusBox.style.color = "var(--text-muted)"; statusBox.innerText = "Sedang menyimpan...";


      fetch(API_URL, {

        method: "POST",

        headers: { "Content-Type": "text/plain;charset=utf-8" },

        body: JSON.stringify({ action: "updatePin", name: currentLockedStaff, oldPin: oldPin, newPin: newPin })

      }).then(res => res.json()).then(res => {

        if(res.success) { statusBox.style.color = "var(--accent-green)"; statusBox.innerText = "✅ Berhasil diperbarui!"; document.getElementById("oldPin").value = ""; document.getElementById("newPin").value = ""; } 

        else { statusBox.style.color = "var(--accent-red)"; statusBox.innerText = "❌ " + res.message; }

    }
